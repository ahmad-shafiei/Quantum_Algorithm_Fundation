\chapter{جعبه ابزار دیجیتال: \lr{Trotter}، \lr{LCU}، \lr{QSP}}
\label{chap:digital-tools}

\section{جایگاه در نقشه}

زیرِ رهیافت تحول زمانی / $f(H)$ (فصل~\ref{chap:evo-adi}، بخش~\ref{sec:time-evolution}). این‌ها الگوریتم کامل «حل هابارد» نیستند؛ زیرروال ساختن یک تابع از هامیلتونی‌اند. خروجی‌شان مدار است.

\section{ایده شهودی}

روی رایانه دیجیتال نمی‌توان $e^{-iHt}$ را به‌صورت یک دکمه داشت. باید آن را از گیت‌های ساده ساخت. سه نسل شهودی: \lr{Trotter} هامیلتونی را تکه‌تکه اعمال می‌کند؛ \lr{LCU} اگر $H$ جمع چند یکانی باشد آن جمع را با کنترل و پس‌گزینش می‌سازد؛ \lr{QSP} وقتی یک بلوکِ کدگذاری‌شده از $H$ دارید تقریباً هر چندجمله‌ای مناسب از آن را با دنباله دوران می‌سازد. هر سه پیاده‌کننده $f(H)$ هستند. تحول زمانی $f(x)=e^{-itx}$ فقط یک انتخاب است~\cite{Childs2018,Berry2015,Low2017}.

در این فصل هر سه ابزار بسط داده می‌شوند: \lr{Trotter} با مثال عددی، سپس \lr{LCU} و \lr{QSP} با فرمول، پارامتر، و مثال کاری.

\section{\lr{Trotter}: فرمول، خطا، پارامترها}
\label{sec:trotter}

فرض کنید
\begin{equation}
	H=\sum_{k=1}^{m} H_k
\end{equation}
و برای هر قطعه، $e^{-iH_k\Delta t}$ به‌آسانی با گیت پیاده می‌شود (مثلاً هر $H_k$ یک جمله پائولی محلی است). اگر جمله‌ها جابه‌جا شوند، $[H_j,H_k]=0$، آن‌گاه
\begin{equation}
	e^{-iHt}=\prod_k e^{-iH_k t}
\end{equation}
دقیق است و تروتر لازم نیست. وقتی جابه‌جا نمی‌شوند، فرمول حاصل‌ضربیِ مرتبه اول می‌گوید~\cite{Childs2018}
\begin{equation}
	\label{eq:trotter1}
	e^{-iHt}
	\approx
	U_{\mathrm{tr}}(t;r)
	:=
	\Biggl(\prod_{k=1}^{m} e^{-iH_k t/r}\Biggr)^{r},
	\qquad
	\Delta t=\frac{t}{r}.
\end{equation}
شهود: در بازه کوتاه $\Delta t$، ترتیب اعمال تکه‌ها کمتر مهم است؛ با ریزتر کردن گام‌ها ($r$ بزرگ‌تر) خطا کم می‌شود و مدار عمیق‌تر می‌گردد. اسکلت همین شکستن در شکل~\ref{fig:trotter-product} آمده است.

\paragraph{خطای تروتر (بیان کاری).}
برای دو جمله $H=A+B$، یک کران کافیِ رایج برای مرتبه اول این است که خطای عملگری مثل
\begin{equation}
	\label{eq:trotter-err}
	\bigl\lVert e^{-i(A+B)t}-U_{\mathrm{tr}}(t;r)\bigr\rVert
	=
	O\!\left(\frac{t^{2}}{r}\,\bigl\lVert[A,B]\bigr\rVert\right)
\end{equation}
رفتار کند (ثابت‌های دقیق به قرارداد هنجار و مرتبه فرمول بستگی دارند). پس برای رسیدن به خطای $\varepsilon$، نوعاً
\begin{equation}
	r \;\gtrsim\; C\,\frac{t^{2}\,\lVert[A,B]\rVert}{\varepsilon}
\end{equation}
کافی است. فرمول‌های مرتبه بالاتر (مثلاً استراتوگ) توان $t$ در صورت را بهتر می‌کنند، اما هر گام مدار پیچیده‌تری دارد.

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item $\{H_k\}$: تجزیه هامیلتونی به قطعات آسان.
	\item $t$: زمان هدفِ شبیه‌سازی.
	\item $r$: تعداد گام‌های تروتر؛ گلوگاه عمق مدار.
	\item $\Delta t=t/r$: طول هر ریزگام.
	\item $\varepsilon$: خطای مجاز در $U$ یا در مشاهده‌پذیر نهایی.
	\item $\lVert[H_j,H_k]\rVert$: شدت ناسازگاری جمله‌ها؛ اگر صفر باشد، $r=1$ کافی است.
\end{itemize}

\section{مثال کاری \lr{Trotter}: یک کیوبیت با $H=X+Z$}
\label{sec:trotter-example}

ادامه مستقیمِ گام ۵ در بخش~\ref{sec:evo-example}: هامیلتونی‌ای که جمله‌هایش جابه‌جا نمی‌شوند، تا تروتر معنا پیدا کند.

\paragraph{گام ۱ --- تعریف مسئله.}
\begin{equation}
	H=X+Z,
	\qquad
	\lvert\psi_0\rangle=\lvert0\rangle,
	\qquad
	t=1,
	\qquad
	O=Z.
\end{equation}
چون $[X,Z]=2iY\neq 0$، نمی‌توان نوشت $e^{-iHt}=e^{-iXt}e^{-iZt}$. ویژه‌مقادیر $H$ برابر $\pm\sqrt{2}$اند، پس $\|H\|_2=\sqrt{2}$.

\paragraph{گام ۲ --- تحول دقیق (مرجع).}
با قطری‌سازی $2\times2$،
\begin{equation}
	U(t)=e^{-i(X+Z)t}
\end{equation}
را می‌سازیم. برای $t=1$ و حالت اولیه $\lvert0\rangle$:
\begin{equation}
	\langle Z\rangle_{\mathrm{exact}}(1)\approx 0.0243.
\end{equation}
این عدد مرجع کلاسیکِ بخش راستی‌آزمایی است: هر مدار تروتر باید به آن نزدیک شود.

\paragraph{گام ۳ --- تقریب حاصل‌ضربی مرتبه اول.}
\begin{equation}
	U_{\mathrm{tr}}(t;r)
	=
	\Bigl(e^{-iX\,t/r}\,e^{-iZ\,t/r}\Bigr)^{r}.
\end{equation}
هر عامل، یک دوران تک‌کیوبیتی است: $e^{-iZ\theta}$ همان $R_Z(2\theta)$ و $e^{-iX\theta}$ همان $R_X(2\theta)$ در قراردادهای رایج مدار است. پس کل مدار فقط دنباله‌ای از $2r$ دوران است.

\paragraph{گام ۴ --- جدول خطا بر حسب $r$.}
جدول~\ref{tab:trotter-error} خطای عملگری $\|U_{\mathrm{tr}}-U\|_2$ و خطای مشاهده‌پذیر $|\langle Z\rangle_{\mathrm{tr}}-\langle Z\rangle_{\mathrm{exact}}|$ را برای $t=1$ نشان می‌دهد. با دو برابر شدن $r$، خطا تقریباً نصف می‌شود؛ هم‌خوان با مقیاس $O(t^{2}/r)$ در~\eqref{eq:trotter-err}.

\paragraph{گام ۵ --- پیوند با بقیه سند.}
\begin{itemize}
	\item همین الگو برای $H=\sum_k H_k$ روی $n$ کیوبیت تکرار می‌شود؛ فقط تعداد جمله‌ها و همبندی مدار زیاد می‌شود.
	\item مسیر آدیاباتیکِ دیجیتال همان تروتر است، اما روی $H(s_j)$ در هر لایه زمانی (فصل~\ref{chap:evo-adi}).
	\item اگر به‌جای $e^{-iHt}$ یک $f(H)$ عمومی بخواهید، نسل بعدی ابزارها (\lr{LCU}، \lr{QSP}) مناسب‌ترند.
\end{itemize}

\section{\lr{LCU}: فرمول و پارامترها}
\label{sec:lcu}

فرض کنید عملگر هدف ترکیب خطی چند یکانی است:
\begin{equation}
	\label{eq:lcu}
	L = \sum_{i=0}^{M-1} \alpha_i U_i,
	\qquad
	\alpha_i\ge 0,
	\qquad
	\Lambda:=\sum_i\alpha_i.
\end{equation}
(اگر بعضی ضرایب منفی یا مختلط باشند، فاز را داخل تعریف $U_i$ جذب می‌کنند.) ایده \lr{LCU} این است که $L/\Lambda$ را به‌عنوان یک \emph{بلوک} داخل یک مدار یکانیِ بزرگ‌تر بنشانید (\emph{بلوک‌کدگذاری}\footnote{\lr{block-encoding}.})~\cite{Berry2015}.

اسکلت استاندارد سه بخش دارد (شکل~\ref{fig:lcu-flow}):
\begin{enumerate}
	\item \textbf{\lr{PREPARE}:} روی ثبات کمکی حالت
	\begin{equation}
		\lvert L\rangle
		=
		\frac{1}{\sqrt{\Lambda}}\sum_i\sqrt{\alpha_i}\,\lvert i\rangle
	\end{equation}
	را بسازید.
	\item \textbf{\lr{SELECT}:} عمل کنترل‌شده
	\begin{equation}
		\lvert i\rangle\lvert\psi\rangle \;\longmapsto\; \lvert i\rangle\,U_i\lvert\psi\rangle.
	\end{equation}
	\item \textbf{پس‌گزینش}\footnote{\lr{postselection}.}: \lr{PREPARE}$^\dagger$ را اعمال کنید و ثبات کمکی را روی $\lvert0\rangle$ اندازه بگیرید. به‌شرط موفقیت، روی سامانه تقریباً $(L/\Lambda)\lvert\psi\rangle$ اعمال شده است.
\end{enumerate}

		\setstretch{1}
		\resizebox{\ifdim\width>\textwidth\textwidth\else\width\fi}{!}{%

		}

	\caption{اسکلت \lr{LCU}. احتمال موفقیت پس‌گزینش به $\lVert L\lvert\psi\rangle\rVert^2/\Lambda^2$ وابسته است؛ برای نزدیک کردن احتمال به یک از تکنیک‌هایی مثل تقویت دامنه استفاده می‌شود.}
	\label{fig:lcu-flow}
\end{figure}

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item $\{U_i,\alpha_i\}$ و $\Lambda=\sum\alpha_i$: تجزیه و عامل نرمال‌سازی.
	\item کیوبیت‌های کمکی: حدود $\lceil\log_2 M\rceil$ برای اندیس $i$.
	\item احتمال پس‌گزینش: گلوگاه تکرار مدار.
	\item خروجی: بلوک‌کدگذاریِ $L/\Lambda$؛ ورودی طبیعی برای \lr{QSP}/\lr{QSVT}.
\end{itemize}

\section{مثال کاری \lr{LCU}: نشاندن $H=X+Z$}
\label{sec:lcu-example}

همان هامیلتونی مثال تروتر را این‌بار به‌عنوان ترکیب خطی دو یکانی می‌نویسیم.

\paragraph{گام ۱ --- تجزیه.}
$X$ و $Z$ خود یکانی‌اند ($X^2=Z^2=I$). پس
\begin{equation}
	H=X+Z,
	\qquad
	\alpha_0=\alpha_1=1,
	\qquad
	\Lambda=2,
	\qquad
	\frac{H}{\Lambda}=\frac{X+Z}{2}.
\end{equation}

\paragraph{گام ۲ --- مدار روی یک کمکی.}
\begin{align}
	\lvert L\rangle
	&=
	\frac{\lvert0\rangle+\lvert1\rangle}{\sqrt{2}}=\lvert+\rangle,
	\\
	\mathrm{SELECT}
	&=
	\lvert0\rangle\langle0\rvert\otimes X
	+
	\lvert1\rangle\langle1\rvert\otimes Z.
\end{align}
یعنی اگر کمکی $0$ باشد $X$ و اگر $1$ باشد $Z$ روی سامانه اعمال می‌شود. سپس هادامارد دوباره روی کمکی (\lr{PREPARE}$^\dagger$) و اندازه‌گیری روی $\lvert0\rangle$.

\paragraph{گام ۳ --- اثر روی $\lvert0\rangle$.}
اگر پس‌گزینش موفق باشد، سامانه به
\begin{equation}
	\frac{H}{2}\lvert0\rangle
	=
	\frac{X+Z}{2}\lvert0\rangle
	=
	\frac{\lvert1\rangle+\lvert0\rangle}{2}
	=
	\frac{1}{\sqrt{2}}\lvert+\rangle
	\cdot \frac{1}{\sqrt{2}}
\end{equation}
می‌رود (تا به هنجار؛ بردار خام $(1/2,1/2)$ است). احتمال موفقیت برای این ورودی
\begin{equation}
	p_{\mathrm{succ}}
	=
	\biggl\lVert\frac{H}{2}\lvert0\rangle\biggr\rVert^2
	=
	\frac12.
\end{equation}
پس در میانگین دو بار اجرا، یک بار بلوکِ مطلوب را می‌گیرید --- بهای کیوبیت کمکی و پس‌گزینش در برابر تروتر.

\paragraph{گام ۴ --- جای این ابزار در نقشه.}
\lr{LCU} خودش لزوماً $e^{-iHt}$ نیست؛ $H$ (یا هر $L$) را داخل یک یکانی بزرگ می‌نشاند. برای دینامیک، معمولاً این بلوک را به \lr{QSP} یا سری تیلور می‌سپارند. برای تروتر، مستقیم $e^{-iX\Delta t}e^{-iZ\Delta t}$ می‌ساختید بدون کمکی.

\section{\lr{QSP}/\lr{QSVT}: فرمول و پارامترها}
\label{sec:qsp}

فرض کنید یک بلوک‌کدگذاری از هامیلتونیِ نرمال‌شده دارید: یکانیِ بزرگ $U$ چنان که در زیر فضای کمکیِ $\lvert0\rangle$،
\begin{equation}
	\label{eq:block-enc}
	\bigl(\langle0\rvert\otimes I\bigr)\,U\,\bigl(\lvert0\rangle\otimes I\bigr)
	=
	\frac{H}{\alpha},
	\qquad
	\biggl\lVert\frac{H}{\alpha}\biggr\rVert\le 1.
\end{equation}
پردازش سیگنال کوانتومی (\lr{QSP}) با یک دنباله فازهای $\phi_0,\ldots,\phi_d$ تقریباً هر چندجمله‌ایِ مناسب $f$ را روی آن بلوک پیاده می‌کند~\cite{Low2017}:
\begin{equation}
	f\!\left(\frac{H}{\alpha}\right)
	\approx
	\text{(بلوک خروجی پس از مدار فازها)}.
\end{equation}
\lr{QSVT} همین ایده را از ویژه‌مقدار به مقادیر تکین تعمیم می‌دهد. کیوبیتیزاسیون همسایه طبیعی است: از روی بلوک‌کدگذاری یک اپراتور پیاده‌روی می‌سازد که ورودی استاندارد \lr{QSP} است.

شکل~\ref{fig:qsp-flow} زنجیره شهودی را نشان می‌دهد: بلوک‌کدگذاری (مثلاً از \lr{LCU}) $\to$ فازهای \lr{QSP} $\to$ تابع $f(H)$.

\paragraph{هزینه شهودی.}
درجه چندجمله‌ای $d$ نوعاً مثل $O\!\left(t\alpha+\log(1/\varepsilon)\right)$ برای تقریب $e^{-iHt}$ رشد می‌کند --- اغلب بهتر از مقیاس تروترِ ساده در رژیم تحمل‌پذیر در برابر خطا. بهایش: نیاز به بلوک‌کدگذاریِ تمیز و طراحی/محاسبه فازها.

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item بلوک‌کدگذاری $H/\alpha$ و عامل $\alpha$ (مثلاً $\Lambda$ در \lr{LCU}).
	\item تابع هدف $f$ (تحول، فیلتر، وارون، \ldots).
	\item درجه $d$ و فازهای $\{\phi_k\}$.
	\item دقت $\varepsilon$ روی بازه $[-1,1]$.
\end{itemize}

\section{مثال کاری \lr{QSP}: از بلوک $H$ تا $f(H)$}
\label{sec:qsp-example}

این مثال عمداً عددیِ سنگین نیست؛ هدف، پر کردن نمادها روی همان $H=X+Z$ است.

\paragraph{گام ۱ --- بلوک آماده.}
از بخش~\ref{sec:lcu-example}، \lr{LCU} بلوک
\begin{equation}
	\frac{H}{\alpha}=\frac{X+Z}{2},
	\qquad
	\alpha=2,
\end{equation}
را می‌سازد. ویژه‌مقادیر $H$ برابر $\pm\sqrt{2}$اند، پس ویژه‌مقادیر بلوک $\pm\sqrt{2}/2\approx\pm 0.707$ داخل $[-1,1]$اند --- شرط~\eqref{eq:block-enc} برقرار است.

\paragraph{گام ۲ --- انتخاب تابع.}
برای شبیه‌سازی در زمان $t$،
\begin{equation}
	f(x)\approx e^{-i(\alpha t)\,x}=e^{-i 2t\, x},
	\qquad
	x\in[-1,1].
\end{equation}
آن‌گاه روی ویژه‌فضای $H$ داریم $f(H/\alpha)\approx e^{-iHt}$. برای فیلتر طیفی یا وارون، $f$ عوض می‌شود؛ مدار بیرونی \lr{QSP} همان الگو را نگه می‌دارد.

\paragraph{گام ۳ --- درجه و دقت.}
به‌صورت کاری، برای خطای $\varepsilon$ در تقریب یکنواخت روی $[-1,1]$، درجه لازم از مرتبه
\begin{equation}
	d \sim O\bigl(\alpha t + \log(1/\varepsilon)\bigr)
\end{equation}
است. مثال عددی کوچک: اگر $\alpha t \sim 1$ و $\varepsilon\sim 10^{-3}$، درجه از مرتبه چند ده کافی است --- در برابر تروتر که برای همان دقت ممکن است $r$ بزرگ‌تری بخواهد (جدول~\ref{tab:trotter-error}).

\paragraph{گام ۴ --- تفسیر.}
\begin{itemize}
	\item \lr{Trotter}: مستقیم $e^{-iH_k\Delta t}$ را می‌چیند؛ ساده، بدون کمکیِ بلوک.
	\item \lr{LCU}: $H$ را بلوک می‌کند؛ خودش $f$ نیست.
	\item \lr{QSP}: روی آن بلوک تقریباً هر $f$ مناسب را می‌سازد؛ زبان مشترک شبیه‌سازی پیشرفته و بسیاری از زیروال‌های \lr{FTQC}.
\end{itemize}

\section{جایگاه در نقشه}

زیرِ پرسش ایستا / طیفی (فصل~\ref{chap:dyn-spec}). \lr{QPE} روی تقاطع با رهیافت تحول می‌نشیند: به $U$ کنترل‌شده نیاز دارد، پس از جعبه ابزار دیجیتال (فصل~\ref{chap:digital-tools}) تغذیه می‌شود. \lr{VQE} رهیافت وردشی / ترکیبی است و هم‌خانواده \lr{Trotter} نیست. \lr{QAOA} خویشاوند ساختاریِ آدیاباتیک و \lr{VQE} برای بهینه‌سازی \lr{Ising} است و در بخش~\ref{sec:qaoa} بسط داده شده است.

\section{ایده شهودی}

دو راه خیلی متفاوت برای انرژی یا طیف، در دو رژیم سخت‌افزاری متفاوت.

\paragraph{\lr{QPE} --- برآورد فاز.}
اگر یکانی $U$ ویژه‌بردار $\lvert\phi\rangle$ با ویژه‌مقدار $e^{2\pi i\varphi}$ داشته باشد، برآورد فاز رقم‌های $\varphi$ را روی کیوبیت‌های کمکی می‌نویسد. با گرفتن $U=e^{-iHt}$، $\varphi$ به ویژه‌مقدار $H$ ترجمه می‌شود. پیش‌نیاز سخت: توان‌های کنترل‌شده $U^{2^k}$ --- یعنی همان شبیه‌سازی فصل~\ref{chap:digital-tools}. به همین دلیل \lr{QPE} الگوریتم طیفیِ خالص نیست. رژیم نوعی: تحمل‌پذیر در برابر خطا، دقت قابل تنظیم، هزینه زیاد مدار~\cite{NielsenChuang2010,McArdle2020}.

\paragraph{\lr{VQE} --- الگوریتم وردشی.}
یک مدار پارامتری $\lvert\psi(\theta)\rangle$ آماده می‌شود، $\langle H\rangle$ با اندازه‌گیری برآورد می‌گردد، و یک بهینه‌ساز کلاسیک $\theta$ را عوض می‌کند. حلقه ترکیبی است: کوانتوم برای حالت و مشاهده‌پذیر، کلاسیک برای جستجوی پارامتر~\cite{Peruzzo2014,Tilly2022}. شهود: به‌جای خواندن طیف با ساعت دقیق، یک آنزات محدود می‌گذارید و انرژی را پایین می‌کشید. مناسب سخت‌افزار نزدیک‌مدت؛ تضمین همگرایی به پایه واقعی ندارد.

جدول~\ref{tab:qpe-vqe} دو رژیم را کنار هم می‌گذارد. سرکوب خطا و تصحیح خطا دو لایه جدا روی این جدول‌اند: اولی غالباً همراه \lr{VQE} و دستگاه نوفه‌دار؛ دومی پیش‌فرض \lr{QPE} مقیاس‌دار~\cite{Preskill2018,Endo2021}.

\section{\lr{QPE}: فرمول و پارامترها}
\label{sec:qpe}

فرض کنید یکانی $U$ ویژه‌بردار $\lvert\phi\rangle$ دارد:
\begin{equation}
	\label{eq:qpe-def}
	U\lvert\phi\rangle = e^{2\pi i\,\varphi}\lvert\phi\rangle,
	\qquad
	\varphi\in[0,1).
\end{equation}
برآورد فاز، با $m$ کیوبیت کمکی (\emph{ثبات شمارش}\footnote{\lr{counting register}.})، تقریبی دودویی از $\varphi$ می‌سازد. اسکلت مدار در شکل~\ref{fig:qpe-flow} آمده است: آماده‌سازی $\lvert\phi\rangle$، اعمال کنترل‌شده $U^{2^{k}}$، تبدیل فوریه وارون (\lr{IQFT}) روی ثبات شمارش، و خوانش~$\tilde\varphi$.

\begin{figure}[htbp]
	\centering
	\begin{latin}
		\setstretch{1}
		\resizebox{\ifdim\width>\textwidth\textwidth\else\width\fi}{!}{%
			\input{figures/fig_qpe_flow.tex}%
		}
	\end{latin}
	\caption{اسکلت \lr{QPE}. هزینه غالب معمولاً ساخت کنترل‌شده $U^{2^{k}}$ است، نه خودِ \lr{IQFT}.}
	\label{fig:qpe-flow}
\end{figure}

\paragraph{از فاز به انرژی.}
در شبیه‌سازی، انتخاب استاندارد $U=e^{-iHt}$ است. اگر $H\lvert\phi\rangle=E\lvert\phi\rangle$، آن‌گاه
\begin{equation}
	e^{-iEt}=e^{2\pi i\,\varphi}
	\quad\Rightarrow\quad
	E = -\frac{2\pi\,\varphi}{t}
	\quad\text{(به پیمانه $2\pi/t$).}
\end{equation}
پس دقت انرژی با دقت فاز و با انتخاب $t$ گره خورده است. $t$ را طوری می‌گیرند که $\varphi$ در $[0,1)$ بدون ابهامِ پیمانه‌ایِ مزاحم خوانده شود (یا بازه انرژی از پیش محدود شود).

\paragraph{دقت و منابع.}
با $m$ بیت شمارش، دقت فاز نوعاً از مرتبه $2^{-m}$ است. برای رسیدن به خطای انرژی $\varepsilon$، هم $m$ و هم کیفیت شبیه‌سازیِ $U^{2^{k}}$ باید با $\varepsilon$ سازگار باشند. گلوگاه عملی اغلب همان شبیه‌سازی کنترل‌شده است (فصل~\ref{chap:digital-tools})، نه شمارش بیت‌ها.

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item $U$ یا جفت $(H,t)$ با $U=e^{-iHt}$.
	\item $\lvert\phi\rangle$: ویژه‌بردار هدف (یا تقریبی از آن؛ در غیر این صورت مخلوط فازها خوانده می‌شود).
	\item $m$: تعداد کیوبیت‌های شمارش؛ دقت $\sim 2^{-m}$.
	\item $\varphi$ و $\tilde\varphi$: فاز واقعی و برآوردشده.
	\item $E$ و $\varepsilon$: انرژی هدف و خطای مجاز.
\end{itemize}

\section{رده مسائل مناسب \lr{QPE}}

\begin{enumerate}
	\item \textbf{انرژی حالت پایه / ویژه‌مقدار با دقت کنترل‌شده} وقتی تقریبی از ویژه‌بردار در دسترس است (یا می‌توان آن را آماده‌سازی کرد).
	\item \textbf{شیمی کوانتومی و مواد در افق \lr{FTQC}}، جایی که عمق مدار زیاد قابل‌قبول است~\cite{McArdle2020}.
	\item \textbf{هر مسئله‌ای که به تخمین فاز یکانی منجر شود} (نه فقط هامیلتونی فیزیکی).
\end{enumerate}
اگر دستگاه کم‌عمق و نوفه‌دار است، یا فقط یک کران بالای انرژی می‌خواهید، \lr{VQE} (بخش~\ref{sec:vqe}) عملی‌تر است. اگر هدف آماده‌سازی پایه از مسیر $H(s)$ است، آدیاباتیک (فصل~\ref{chap:evo-adi}) را جدا ببینید --- \lr{QPE} از آدیاباتیک تغذیه نمی‌شود؛ از $U$ تغذیه می‌شود.

\section{مثال کاری \lr{QPE}: یک کیوبیت با فاز $\varphi=1/4$}
\label{sec:qpe-example}

هدف: همه نمادهای بخش~\ref{sec:qpe} را روی کوچک‌ترین نمونه دقیق پر کنیم، طوری که خوانش دودویی بدون خطای گرد کردن باشد.

\paragraph{گام ۱ --- یکانی و ویژه‌بردار.}
کیوبیت سامانه را در $\lvert1\rangle$ می‌گذاریم و
\begin{equation}
	U
	=
	\begin{pmatrix}
		1&0\\
		0&e^{2\pi i/4}
	\end{pmatrix}
	=
	\begin{pmatrix}
		1&0\\
		0&i
	\end{pmatrix}
\end{equation}
را می‌گیریم. آن‌گاه
\begin{equation}
	U\lvert1\rangle = e^{2\pi i\cdot(1/4)}\lvert1\rangle,
	\qquad
	\varphi=\frac14=0.01_{\mathrm{bin}}.
\end{equation}

\paragraph{گام ۲ --- پیوند با هامیلتونی.}
همین $U$ از تحول $H=Z$ با زمان $t=\pi/2$ روی ویژه‌حالت $\lvert1\rangle$ می‌آید:
\begin{equation}
	e^{-iZ(\pi/2)}\lvert1\rangle
	=
	e^{i\pi/2}\lvert1\rangle
	=
	i\lvert1\rangle,
\end{equation}
که با قرارداد~\eqref{eq:qpe-def} همان $\varphi=1/4$ است. از فرمول انرژی،
\begin{equation}
	E = -\frac{2\pi\cdot(1/4)}{\pi/2} = -1,
\end{equation}
و واقعاً $Z\lvert1\rangle=-\lvert1\rangle$. پس خواندن فاز، خواندن انرژی است.

\paragraph{گام ۳ --- ثبات شمارش با $m=2$.}
دو کیوبیت کمکی با هادامارد آماده می‌شوند؛ سپس $U^{2^{0}}$ و $U^{2^{1}}$ کنترل‌شده اعمال می‌گردند و \lr{IQFT} روی کمکی‌ها اجرا می‌شود. چون $\varphi=1/4$ دقیقاً با دو بیت نمایش‌پذیر است، اندازه‌گیری ثبات شمارش با احتمال یک، بیت‌های $01$ را می‌دهد:
\begin{equation}
	\tilde\varphi = 0.01_{\mathrm{bin}} = \frac14=\varphi.
\end{equation}
اگر $m=1$ بود، دقت کافی نبود و احتمال خطا در گرد کردن ظاهر می‌شد --- همان بهای کم کردن کیوبیت شمارش.

\paragraph{گام ۴ --- تفسیر در نقشه سند.}
\begin{itemize}
	\item پرسش: طیفی (ویژه‌مقدار / انرژی).
	\item رهیافت: تحول ($U=e^{-iHt}$) $+$ برآورد فاز.
	\item ابزار سنگین: در این مثال $U$ یک قطریِ ساده است؛ در مدل واقعی همان $U^{2^{k}}$ را \lr{Trotter}/\lr{LCU}/\lr{QSP} می‌سازند.
\end{itemize}

\section{\lr{VQE}: فرمول و پارامترها}
\label{sec:vqe}

به‌جای ساختن ساعتِ فاز، یک خانواده پارامتری از حالت‌ها انتخاب می‌کنید (\emph{آنزات}\footnote{\lr{ansatz}.}):
\begin{equation}
	\lvert\psi(\theta)\rangle = V(\theta)\lvert0\rangle^{\otimes n},
\end{equation}
و انرژی آزمایشی را کمینه می‌کنید:
\begin{equation}
	\label{eq:vqe}
	E(\theta)=\langle\psi(\theta)\rvert H\lvert\psi(\theta)\rangle,
	\qquad
	\theta^\star=\arg\min_\theta E(\theta).
\end{equation}
بنابر اصل وردشی، $E(\theta)\ge E_0$ برای هر $\theta$؛ برابری فقط اگر آنزات بتواند پایه واقعی را بسازد. حلقه در شکل~\ref{fig:vqe-loop} است: مدار پارامتری $\to$ برآورد $\langle H\rangle$ با شات $\to$ بهینه‌ساز کلاسیک $\to$ به‌روزرسانی $\theta$.

\begin{figure}[htbp]
	\centering
	\begin{latin}
		\setstretch{1}
		\resizebox{0.92\textwidth}{!}{%
			\input{figures/fig_vqe_loop.tex}%
		}
	\end{latin}
	\caption{حلقه ترکیبی \lr{VQE}. بخش کوانتومی کم‌عمق است؛ سنگینی کار به تعداد پارامتر، شات، و جمله‌های $H$ برمی‌گردد.}
	\label{fig:vqe-loop}
\end{figure}

\paragraph{اندازه‌گیری $H$.}
معمولاً
\begin{equation}
	H=\sum_j c_j P_j
\end{equation}
با $P_j$ حاصل‌ضرب پائولی. هر $P_j$ جداگانه اندازه‌گیری می‌شود و
\begin{equation}
	\widehat{E(\theta)}=\sum_j c_j\,\widehat{\langle P_j\rangle}.
\end{equation}
تعداد شات، واریانس برآورد را کنترل می‌کند؛ این هزینه کنار عمق مدار می‌نشیند.

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item $H=\sum c_j P_j$: هامیلتونی و تجزیه اندازه‌گیری.
	\item $V(\theta)$ / آنزات: خانواده حالت‌های قابل ساخت با مدار کم‌عمق.
	\item $\theta$: بردار پارامتر؛ ابعادش پیچیدگی بهینه‌سازی را می‌سازد.
	\item $E(\theta)$ و $E_0$: انرژی آزمایشی و انرژی پایه واقعی.
	\item تعداد شات، و انتخاب بهینه‌ساز کلاسیک.
\end{itemize}

\section{رده مسائل مناسب \lr{VQE}}

\begin{enumerate}
	\item \textbf{تقریب حالت پایه روی \lr{NISQ}} وقتی عمق \lr{QPE} در دسترس نیست~\cite{Peruzzo2014,Tilly2022}.
	\item \textbf{شیمی و اسپین در مقیاس کوچک تا متوسط} با آنزات مسئله‌محور (مثلاً \lr{UCC} در شیمی، یا لایه‌های سخت‌افزاری‌پسند).
	\item \textbf{کاوش اولیه} قبل از سرمایه‌گذاری روی شبیه‌سازی سنگینِ $U$ برای \lr{QPE}.
\end{enumerate}
محدودیت اصلی: مینیمم به‌دست‌آمده ممکن است مینیمم محلی یا سقفِ آنزات باشد، نه $E_0$. برخلاف \lr{QPE}، افزایش صرفِ «دقت ساعت» وجود ندارد؛ باید آنزات یا بهینه‌سازی را عوض کرد.

\section{مثال کاری \lr{VQE}: یک کیوبیت با آنزات $R_Y$}
\label{sec:vqe-example}

ادامه همان خانواده تک‌کیوبیتی فصل‌های تحول و تروتر، اما این‌بار پرسش طیفی است.

\paragraph{گام ۱ --- هامیلتونی.}
\begin{equation}
	H = \tfrac12 Z + \tfrac14 X.
\end{equation}
طیف دقیق با قطری‌سازی:
\begin{equation}
	E_0 = -\sqrt{\bigl(\tfrac12\bigr)^2+\bigl(\tfrac14\bigr)^2}
	= -\frac{\sqrt5}{4}
	\approx -0.5590,
	\qquad
	E_1=+0.5590.
\end{equation}

\paragraph{گام ۲ --- آنزات.}
\begin{equation}
	\lvert\psi(\theta)\rangle
	=
	R_Y(\theta)\lvert0\rangle
	=
	\cos\frac\theta2\,\lvert0\rangle
	+
	\sin\frac\theta2\,\lvert1\rangle.
\end{equation}
آن‌گاه به‌صورت تحلیلی
\begin{equation}
	E(\theta)
	=
	\tfrac12\cos\theta + \tfrac14\sin\theta.
\end{equation}
کمینه وقتی مشتق صفر شود به دست می‌آید و مقدار کمینه همان $- \sqrt{5}/4$ است: این آنزات تک‌پارامتری برای یک کیوبیت \emph{کامل} است و به پایه واقعی می‌رسد.

\paragraph{گام ۳ --- جدول انرژی آزمایشی.}
جدول~\ref{tab:vqe-scan} چند نمونه از روبش $\theta$ را نشان می‌دهد. حول $\theta\approx 3.61$ انرژی به $E_0$ می‌رسد.

\end{table}

\paragraph{گام ۴ --- اندازه‌گیری عملی.}
روی دستگاه، $\langle Z\rangle$ و $\langle X\rangle$ جداگانه با شات برآورد می‌شوند و
\begin{equation}
	\widehat{E}=\tfrac12\widehat{\langle Z\rangle}+\tfrac14\widehat{\langle X\rangle}.
\end{equation}
بهینه‌ساز کلاسیک $\theta$ را طوری جلو می‌برد که $\widehat{E}$ کم شود. در نبود نوفه و با شات کافی، همان مینیمم جدول بازتولید می‌شود.

\paragraph{گام ۵ --- تفسیر در نقشه سند.}
\begin{itemize}
	\item پرسش: طیفی (حالت پایه)، اما روش وردشی است نه برآورد فاز.
	\item رهیافت: ترکیبی؛ مدار کم‌عمق $+$ کلاسیک.
	\item تفاوت با آدیاباتیک: اینجا مسیر $H(s)$ نیست؛ فضای پارامترِ آنزات است.
	\item تفاوت با \lr{QPE}: به $U^{2^{k}}$ نیاز نیست؛ دقت را آنزات و بهینه‌سازی محدود می‌کنند.
\end{itemize}

\section{\lr{QAOA}: فرمول، شهود، و جای آن در نقشه}
\label{sec:qaoa}

\lr{QAOA} (\lr{Quantum Approximate Optimization Algorithm}) یک الگوریتم بهینه‌سازی تقریبی کوانتومی است~\cite{Farhi2014}. هدف نوعی‌اش یافتن پیکربندی کم‌انرژی برای یک هامیلتونی کلاسیکِ قطری در پایه محاسباتی است --- همان زبان \lr{Ising} بخش~\ref{sec:ising} --- نه خواندن طیف با دقت دلخواه مثل \lr{QPE}.

\paragraph{جایگاه در سه لایه.}
\begin{itemize}
	\item \textbf{پرسش:} بهینه‌سازی ترکیبی / حالت پایه $H_C$ (مثلاً برش بیشینه، پوشش، اسپین‌شیشه).
	\item \textbf{رهیافت:} پل میان محاسبه آدیاباتیک و وردشیِ ترکیبی.
	\item \textbf{ابزار اجرا:} مدار کم‌عمقِ لایه‌لایه روی \lr{NISQ}، به‌علاوه بهینه‌ساز کلاسیک برای زاویه‌ها.
\end{itemize}

\paragraph{ایده شهودی در یک بند.}
از $\lvert+\rangle^{\otimes n}$ شروع می‌کنید (پایه آسانِ مخلوط‌کننده). یک لایه «هزینه» فازهایی متناسب با انرژی کلاسیک روی هر رشته بیتی می‌نویسد؛ یک لایه «مخلوط‌کننده» رشته‌ها را با میدان عرضی جابه‌جا می‌کند. این جفت را $p$ بار تکرار می‌کنید و زاویه‌ها را کلاسیک طوری تنظیم می‌کنید که $\langle H_C\rangle$ کم شود. شکل~\ref{fig:qaoa-layers} همین تناوب را نشان می‌دهد.

\subsection{فرمول و پارامترها}

هامیلتونی \emph{مسئله} (هزینه) معمولاً قطری است:
\begin{equation}
	H_C = \sum_{i<j} J_{ij} Z_i Z_j + \sum_i h_i Z_i
\end{equation}
(همان~\eqref{eq:ising-classical} با قرارداد مناسب). مخلوط‌کننده استاندارد میدان عرضی یکنواخت است:
\begin{equation}
	H_B = \sum_{i=1}^{n} X_i.
\end{equation}
حالت $p$ لایه:
\begin{equation}
	\label{eq:qaoa}
	\lvert\vec\gamma,\vec\beta\rangle
	=
	\Biggl(\prod_{j=1}^{p}
		e^{-i\beta_j H_B}\,
		e^{-i\gamma_j H_C}
	\Biggr)
	\lvert+\rangle^{\otimes n}.
\end{equation}
تابع هزینه آزمایشی
\begin{equation}
	E(\vec\gamma,\vec\beta)
	=
	\langle\vec\gamma,\vec\beta\rvert H_C \lvert\vec\gamma,\vec\beta\rangle
\end{equation}
را با اندازه‌گیری جمله‌های پائولیِ $H_C$ برمی‌آورید و یک بهینه‌ساز کلاسیک $(\vec\gamma,\vec\beta)$ را عوض می‌کند --- دقیقاً حلقه \lr{VQE}، با این تفاوت که آنزات اینجا ساختار ثابتِ مسئله/مخلوط‌کننده دارد.

\paragraph{پارامترها، یک‌جا.}
\begin{itemize}
	\item $H_C$: تابع هزینه / \lr{Ising} کلاسیک؛ جواب در رشته بیتی پایه آن رمز است.
	\item $H_B$: مخلوط‌کننده؛ معمولاً $\sum X_i$ تا از $\lvert+\rangle^{\otimes n}$ همه پایه را ببیند.
	\item $p$: تعداد لایه‌ها؛ عمق مدار و تعداد پارامترها $2p$ است.
	\item $\gamma_j$: زاویه لایه هزینه (چقدر فازِ $H_C$ اعمال شود).
	\item $\beta_j$: زاویه لایه مخلوط‌کننده.
	\item شات‌ها و بهینه‌ساز کلاسیک: همان هزینه‌های \lr{VQE}.
\end{itemize}

\subsection{پیوند با آدیاباتیک و با \lr{VQE}}

مسیر آدیاباتیک $H(s)=(1-s)H_B+s H_C$ اگر با تروتر در چند تکه درشت گسسته شود، همان تناوب
$e^{-i\Delta t_C H_C}\,e^{-i\Delta t_B H_B}$
ظاهر می‌شود. \lr{QAOA} این $\Delta t$ها را آزاد می‌گذارد تا کلاسیک بهینه کند؛ دیگر لازم نیست $s(t)$ حتماً آهسته و مطابق کران $\Delta_{\min}$ باشد، اما دیگر تضمین آدیاباتیک هم ندارید.

\subsection{رده مسائل مناسب}

\begin{enumerate}
	\item مسائل بهینه‌سازی نگاشته‌شده به \lr{Ising} / \lr{MaxCut}.
	\item وقتی سخت‌افزار عمق کم دارد و می‌خواهید از ساختار $H_C$--$H_B$ استفاده کنید، نه از یک آنزات بی‌ساختار.
	\item مطالعه تقریبی بودن: نسبت تقریب و رفتار با افزایش $p$.
\end{enumerate}
اگر انرژی مولکولی با آنزات شیمی می‌خواهید، \lr{VQE}/\lr{UCC} طبیعی‌تر است. اگر طیف با دقت کنترل‌شده می‌خواهید، \lr{QPE}. اگر گاف مسیر را می‌شناسید و سخت‌افزار مسیر می‌دهد، آدیاباتیک را جدا ببینید.

\section{مثال کاری \lr{QAOA}: دو کیوبیت و $H_C=-Z_1Z_2$}
\label{sec:qaoa-example}

هدف: نمادهای بخش~\ref{sec:qaoa} را روی همان \lr{Ising} دوکیوبیتیِ آشنا پر کنیم و ببینیم یک لایه چگونه نزدیک پایه می‌شود.

\paragraph{گام ۱ --- مسئله به‌زبان هزینه.}
\begin{equation}
	H_C=-Z_1 Z_2,
	\qquad
	H_B=X_1+X_2,
	\qquad
	\lvert\psi_0\rangle=\lvert++\rangle,
	\qquad
	p=1.
\end{equation}
روی پایه محاسباتی:
\begin{equation}
	E_C(\lvert00\rangle)=E_C(\lvert11\rangle)=-1,
	\qquad
	E_C(\lvert01\rangle)=E_C(\lvert10\rangle)=+1.
\end{equation}
پس کمینه کلاسیک $-1$ است (پیکربندی‌های هم‌جهت). میانگین روی $\lvert++\rangle$ صفر است --- نقطه شروعِ «تصادفیِ یکنواخت» روی چهار رشته.

\paragraph{گام ۲ --- یک لایه صریح.}
\begin{equation}
	\lvert\gamma,\beta\rangle
	=
	e^{-i\beta H_B}\,
	e^{-i\gamma H_C}\,
	\lvert++\rangle.
\end{equation}
$e^{-i\gamma H_C}$ چون $H_C$ قطری است، فقط فاز روی هر $\lvert zz'\rangle$ می‌نویسد. $e^{-i\beta H_B}$ همان دوران‌های $X$ روی دو کیوبیت است و دامنه را میان رشته‌ها جابه‌جا می‌کند.

\paragraph{گام ۳ --- روبش و مینیمم.}
با روبش ظریف‌تر $(\gamma,\beta)$، می‌توان به خودِ پایه رسید:
\begin{equation}
	E(\gamma^\star,\beta^\star)=-1
	\quad\text{با}\quad
	\gamma^\star=\pi/4,
	\quad
	\beta^\star\approx 1.18
\end{equation}
(و نقاط معادل دوره‌ای). در این نقطه احتمال اندازه‌گیری روی $\{\lvert00\rangle,\lvert11\rangle\}$ هر کدام $1/2$ است و روی رشته‌های ناهم‌جهت تقریباً صفر --- یعنی هر شات یکی از جواب‌های بهینه را می‌دهد.

\paragraph{گام ۴ --- نسبت تقریب (شهودی).}
اگر خطِ پایه را انرژی شروع $E_{\mathrm{start}}=\langle++\rvert H_C\lvert++\rangle=0$ بگیریم،
\begin{equation}
	r
	=
	\frac{E_{\mathrm{start}}-E(\gamma^\star,\beta^\star)}{E_{\mathrm{start}}-E_0}
	=
	\frac{0-(-1)}{0-(-1)}
	=1.
\end{equation}
روی این مسئله دوکیوبیتی با $p=1$ نسبت تقریب کامل است. روی گراف‌های بزرگ‌تر و سخت‌تر، $r$ با $p=1$ معمولاً کوچک‌تر است و افزایش $p$ موضوع پژوهش می‌شود.

\paragraph{گام ۵ --- اجرا روی دستگاه.}
\begin{enumerate}
	\item دو هادامارد برای $\lvert++\rangle$.
	\item فازهای $ZZ$ مطابق $\gamma$ (یک گیت دوکیوبیتیِ قطری یا معادلش).
	\item دوران‌های $X$ مطابق $\beta$.
	\item تکرار شات، برآورد $\langle Z_1Z_2\rangle$، و به‌روزرسانی $(\gamma,\beta)$ با بهینه‌ساز کلاسیک.
\end{enumerate}

\paragraph{گام ۶ --- تفسیر در نقشه سند.}
\begin{itemize}
	\item نسبت به آدیاباتیک (بخش~\ref{sec:adi-ising-example}): همان $H_C$ و همان $H_B$؛ آنجا مسیر پیوسته و شرط گاف، اینجا زاویه‌های گسسته و بهینه‌سازی.
	\item نسبت به \lr{VQE} (بخش~\ref{sec:vqe-example}): حلقه یکی است؛ آنزات اینجا از فیزیکِ مسئله/مخلوط‌کننده می‌آید نه از یک $R_Y$ آزاد.
	\item نسبت به \lr{QPE}: هدف تقریبِ ترکیبیاتی است، نه رقم‌های فاز با دقت $2^{-m}$.
\end{itemize}

دقت طیفی در افق \lr{FTQC} و وقتی تقریبی از ویژه‌بردار دارید: \lr{QPE} (بخش‌های~\ref{sec:qpe} و~\ref{sec:qpe-example}). آزمایش روی دستگاه امروز یا کاوش اولیه: \lr{VQE} (بخش‌های~\ref{sec:vqe} و~\ref{sec:vqe-example}). بهینه‌سازی \lr{Ising}/برش بیشینه با آنزات لایه‌لایه: \lr{QAOA} (بخش‌های~\ref{sec:qaoa} و~\ref{sec:qaoa-example}). اگر گاف مسیر معلوم و سخت‌افزار آنالوگ/آدیاباتیک دارید: اول فصل~\ref{chap:evo-adi}.

\begin{remark}[جای توسعه]
برآورد فاز تکیه‌گاه‌شده و کیوبیتیزاسیون$+$\lr{QPE}؛ اثر هم‌پوشانی ناقص با ویژه‌بردار؛ \lr{VQD} و آنزات‌های شیمی (\lr{UCC})؛ \lr{QAOA} با $p>1$ و مسائل \lr{MaxCut} بزرگ‌تر؛ مخلوط‌کننده‌های غیر استاندارد؛ نقش تعداد شات؛ چشم‌انداز بایرن (\lr{barren plateau}).
\end{remark}

\chapter{دیجیتال، آنالوگ، ترکیبی}
\label{chap:hardware}

\section{جایگاه در نقشه}

پایین نمودار، بعد از «الگوریتم»: مدار، مسیر آدیاباتیک، یا زمان‌بندی آنالوگ باید روی یک دستگاه بنشیند. این فصل مدل سخت‌افزاری است، نه فهرست شرکت‌ها. شبیه‌سازی آنالوگ چون اغلب با آدیاباتیک و با تحول دیجیتال قاطی می‌شود، در بخش~\ref{sec:analog} جداگانه باز شده است.

\section{ایده شهودی}

\paragraph{دیجیتال --- مدار گیتی.}
الگوریتم به دنباله گیت از یک مجموعه گیت جهانی ترجمه می‌شود. انعطاف زیاد است؛ هزینه به عمق مدار، همبندی، و خطای گیت حساس است. \lr{Trotter}، \lr{LCU}، \lr{QSP}، \lr{QPE} و \lr{VQE} همگی در نهایت اینجا مدار می‌شوند --- \lr{VQE} معمولاً مدار کم‌عمق‌تر~\cite{Preskill2018}.

\paragraph{آنالوگ --- مهندسی هامیلتونی.}
به‌جای شکستن $U(t)$، $H_{\mathrm{hw}}$ را نزدیک $H_{\mathrm{target}}$ می‌سازید و زمان را واقعی جلو می‌برید. مزیت: عمق گیت معنا ندارد. محدودیت: فقط مدل‌هایی که دستگاه بلداست~\cite{CiracZoller2012,Georgescu2014}. جزئیات و مثال در بخش~\ref{sec:analog}.

\paragraph{ترکیبی --- حلقه با کلاسیک.}
بخشی از کار روی پردازنده کوانتومی است، بخشی روی کلاسیک. \lr{VQE} نمونه بارز است. سرکوب خطا هم اغلب ترکیبی است. سخت‌افزار بازپیکربندی‌پذیر مرز آنالوگ و دیجیتال را نرم می‌کند: همبندی را می‌توان با مسئله عوض کرد.

\section{شبیه‌سازی آنالوگ: فرمول و تمایزها}
\label{sec:analog}

\subsection{تعریف کاری}

شبیه‌سازی آنالوگ یعنی سامانه آزمایشگاهی را طوری تنظیم کنید که هامیلتونی‌اش تقریباً همان مدل هدف باشد:
\begin{equation}
	\label{eq:analog-map}
	H_{\mathrm{hw}}(t) \;\approx\; H_{\mathrm{target}}(t),
\end{equation}
سپس بگذارید طبیعت انتگرال زمانی را بگیرد:
\begin{equation}
	i\,\partial_t\lvert\psi\rangle = H_{\mathrm{hw}}(t)\,\lvert\psi\rangle.
\end{equation}
هیچ تجزیه‌ای به $e^{-iH_k\Delta t}$ و گیت سراسری لازم نیست؛ «مدار» همان زمان‌بندیِ کنترل‌های فیزیکی است (شدت لیزر، شار مغناطیسی، فاصله اتم‌ها، \ldots). اسکلت در شکل~\ref{fig:analog-sim} آمده است.

\subsection{سه چیزی که نباید قاطی شوند}

\begin{itemize}
	\item \textbf{آنالوگ در برابر دیجیتال.} دیجیتال: $H$ را به گیت می‌شکند (\lr{Trotter}/\lr{LCU}/\lr{QSP}). آنالوگ: $H$ را روی دستگاه می‌سازد. هر دو می‌توانند همان پرسش دینامیکی را جواب بدهند؛ هزینه و خطایشان فرق دارد.
	\item \textbf{آنالوگ در برابر محاسبه آدیاباتیک.} آدیاباتیک یک \emph{الگوریتم} با مسیر $H(s)$ برای رسیدن به پایه است. آنالوگ یک \emph{بستر اجرا} است. می‌توانید تحولِ $H$ ثابت را آنالوگ کنید، یا مسیر آدیاباتیک را آنالوگ پیاده کنید؛ لزوماً یکی نیستند (فصل~\ref{chap:evo-adi}).
	\item \textbf{آنالوگ در برابر بازپخت کوانتومی.} بازپخت خویشاوند نوفه‌دار و بهینه‌سازی‌محور است؛ هدفش غالباً کمینه تقریبی است، نه شبیه‌سازی دقیق دینامیک مدل فیزیکی.
\end{itemize}

\subsection{پارامترها و گلوگاه‌ها}

\begin{itemize}
	\item $H_{\mathrm{target}}$: مدل پژوهشی (مثلاً \lr{TFIM}، هابارد مهندسی‌شده).
	\item نگاشت: کدام کنترل دستگاه کدام جمله $H$ را می‌سازد ($J\leftrightarrow$ جفت‌شدگی، $h_x\leftrightarrow$ میدان رانش، \ldots).
	\item وفاداری نگاشت: $\lVert H_{\mathrm{hw}}-H_{\mathrm{target}}\rVert$ یا خطای مشاهده‌پذیر نسبت به مدل ایده‌آل.
	\item زمان فیزیکی $t$ (و در صورت نیاز زمان‌بندی $H(t)$).
	\item ناهمدوسی و کالیبراسیون: در آنالوگ «عمق مدار» نیست؛ زمان همدوسی و پایداری کنترل‌ها گلوگاه‌اند.
\end{itemize}

\subsection{رده مسائل مناسب آنالوگ}

\begin{enumerate}
	\item مدل‌هایی که سخت‌افزار ذاتاً بلداست (اسپین با $ZZ$ و میدان، بعضی رژیم‌های هابارد در گازهای سرد، \ldots).
	\item دینامیک در زمان‌های متوسط وقتی تجزیه گیتی خیلی عمیق می‌شود.
	\item کاوش فاز یا quench وقتی دقت «شیمیایی» \lr{FTQC} لازم نیست.
\end{enumerate}
اگر به $f(H)$ دلخواه، یا به کنترل خطای منطقی در مقیاس بزرگ نیاز دارید، مسیر دیجیتال تحمل‌پذیر در برابر خطا مناسب‌تر است.

\section{مثال کاری آنالوگ: دینامیک دو اسپینِ \lr{TFIM}}
\label{sec:analog-example}

همان خانواده مدل که در بخش~\ref{sec:ising} و مثال آدیاباتیک دیدید، این‌بار نه برای یافتن پایه، بلکه برای \emph{تحول زمانی} روی سخت‌افزار آنالوگ.

\paragraph{گام ۱ --- مدل هدف.}
\begin{equation}
	\label{eq:analog-tfim}
	H_{\mathrm{target}}
	=
	-J\, Z_1 Z_2
	-
	h_x\,(X_1+X_2),
	\qquad
	J,h_x>0.
\end{equation}
این یک \lr{TFIM} دوسایته است: جفت‌شدگی فرومغناطیسی به‌علاوه میدان عرضی. پرسش پژوهشی نمونه: اگر سامانه در $\lvert00\rangle$ شروع کند، $\langle Z_1 Z_2\rangle(t)$ چگونه نوسان می‌کند؟

\paragraph{گام ۲ --- نگاشت به سخت‌افزار.}
فرض کنید دستگاه (مثلاً دو یون یا دو کیوبیت ابررسانا در رژیم کوپلینگِ قابل تنظیم) بتواند همزمان
\begin{equation}
	H_{\mathrm{hw}}
	=
	-J_{\mathrm{hw}}\, Z_1 Z_2
	-
	h_{\mathrm{hw}}\,(X_1+X_2)
\end{equation}
را با کنترل‌های آزمایشگاهی بسازد. شبیه‌سازی آنالوگ یعنی کالیبره کردن
\begin{equation}
	J_{\mathrm{hw}}\approx J,
	\qquad
	h_{\mathrm{hw}}\approx h_x,
\end{equation}
نه نوشتن مدار تروتر. اگر دستگاه فقط یکی از جمله‌ها را بلداست، یا باید مدل را عوض کنید یا به دیجیتال برگردید.

\paragraph{گام ۳ --- اجرا.}
\begin{enumerate}
	\item حالت اولیه $\lvert00\rangle$ را آماده کنید (در بسیاری از سکوها حالت طبیعی است).
	\item کنترل‌ها را روی $(J_{\mathrm{hw}},h_{\mathrm{hw}})$ بگذارید و زمان فیزیکی $t$ صبر کنید --- سامانه خودش $e^{-iH_{\mathrm{hw}}t}$ را اعمال می‌کند.
	\item $\langle Z_1 Z_2\rangle$ یا جمعیت پایه‌های محاسباتی را بخوانید.
\end{enumerate}

\paragraph{گام ۴ --- مرجع کلاسیک روی همین دو کیوبیت.}
چون $n=2$ است، $H_{\mathrm{target}}$ را می‌توان دقیقاً قطری کرد و $\langle Z_1 Z_2\rangle_{\mathrm{exact}}(t)$ را ساخت. آزمایش آنالوگ وقتی موفق است که منحنی اندازه‌گیری‌شده به این مرجع نزدیک باشد. انحراف، ترکیبی از خطای نگاشت ($J_{\mathrm{hw}}\neq J$)، نوفه، و خطای خوانش است --- نه «خطای تروتر»، چون تروتری در کار نبوده است.

با انتخاب عددی $J=1$، $h_x=1$ و شروع از $\lvert00\rangle$، مرجع دقیق می‌دهد
\begin{equation}
	\langle Z_1 Z_2\rangle(0)=1,
	\qquad
	\langle Z_1 Z_2\rangle(1)\approx 0.50,
	\qquad
	\langle Z_1 Z_2\rangle(\pi/2)\approx 0.89.
\end{equation}
هم‌بستگی با زمان نوسان می‌کند؛ منحنی کامل همان بنچمارک راستی‌آزمایی برای اجرای آنالوگ است.

\paragraph{گام ۵ --- مقایسه با مسیر دیجیتال روی همان $H$.}
اگر همین $H_{\mathrm{target}}$ را دیجیتال شبیه‌سازی کنید، از بخش~\ref{sec:trotter} باید
\begin{equation}
	e^{-iHt}
	\approx
	\bigl(e^{iJ Z_1Z_2\,t/r}\,e^{i h_x(X_1+X_2)\,t/r}\bigr)^{r}
\end{equation}
را با گیت بسازید و $r$ را برای خطای تروتر بالا ببرید. در آنالوگ، همان فیزیک بدون این حاصل‌ضرب پیاده می‌شود؛ بهایش وابستگی به توانایی سخت‌افزار در ساخت همزمان هر دو نوع جمله است.

\paragraph{گام ۶ --- تفسیر در نقشه سند.}
\begin{itemize}
	\item پرسش: دینامیکی ($\langle Z_1Z_2\rangle(t)$)، نه لزوماً حالت پایه.
	\item رهیافت: تحول زمانی.
	\item اجرا: آنالوگ (مهندسی $H$)، در برابر دیجیتال (تروтер روی همان $H$).
	\item اگر به‌جای $H$ ثابت، کنترل‌ها را آرام از فقط‌عرضی به فقط‌$ZZ$ جارو کنید، همان آزمایش می‌تواند به محاسبه آدیاباتیک آنالوگ تبدیل شود --- تغییر در زمان‌بندی کنترل، نه در تعریف آنالوگ بودن.
\end{itemize}

\section{حداقل فرمالیسم --- چک‌لیست اجرا}

سؤال عملی برای هر پروژه:
\begin{enumerate}
	\item خروجی الگوریتم مدار است، زمان‌بندی $H(t)$ است، یا هر دو؟
	\item دستگاه می‌تواند جمله‌های $H$ را مستقیم بسازد یا فقط گیت جهانی دارد؟
	\item حلقه کلاسیک لازم است (وردشی، کالیبراسیون، سرکوب خطا)؟
\end{enumerate}
اگر پاسخ (۲) مثبت باشد، مسیر آنالوگ (بخش‌های~\ref{sec:analog} و~\ref{sec:analog-example}) را جدی بگیرید. اگر مسئله $f(H)$ عمومی و دقت بالا می‌خواهد، مسیر دیجیتال تحمل‌پذیر در برابر خطا را جدی بگیرید.

\chapter{بستن حلقه: اندازه‌گیری، راستی‌آزمایی، برآورد منابع}
\label{chap:closing}

\section{جایگاه در نقشه}

انتهای جریان کاری. بدون این سه گام، مدار یا مسیر آدیاباتیک فقط یک آزمایش اجراشده است، نه نتیجه پژوهشی قابل دفاع.

\section{ایده شهودی}

\paragraph{از مدار تا مشاهده‌پذیر.}
خروجی رایانه کوانتومی تابع موج کامل نیست. خوانش در یک پایه بیت‌های کلاسیک می‌دهد. مشاهده‌پذیر $O$ معمولاً به جمله‌های قابل اندازه‌گیری شکسته می‌شود و میانگین شات‌ها $\langle O\rangle$ را می‌سازد. وفاداری
\ft{fidelity}
 با یک حالت هدف در سامانه‌های بزرگ مستقیماً در دسترس نیست.

\paragraph{راستی‌آزمایی کلاسیک.}
قبل از باور به عدد، آن را جایی که کلاسیک هنوز قوی است چک کنید: شبکه یا مولکول خیلی کوچک (قطری‌سازی یا شبیه‌سازی بردار حالت)؛ حدهای دقیقاً حل‌پذیر ($U=0$ در هابارد، میدان صفر در اسپین)؛ تقارن‌ها (تعداد ذره، اسپین، بار پیمانه‌ای)؛ شبکه تانسوری در رژیم درهم‌تنیدگی کم~\cite{Daley2022,Cirac2021}. اگر الگوریتم در این حدها غلط است، در حد سخت هم به آن اعتماد نکنید.

\paragraph{برآورد منابع.}
سؤال پژوهشی «آیا این مسیر شدنی است؟»: چند کیوبیت پس از کدگذاری؟ عمق مدار یا زمان آنالوگ چقدر است؟ برای \lr{Trotter} چند گام تا خطای تروتر زیر $\varepsilon$؟ برای \lr{QSP} درجه چندجمله‌ای؟ برای \lr{VQE} تعداد پارامتر، شات، و جمله‌های $H$؟ نویز دستگاه در برابر دقت هدف: سرکوب خطا کافی است یا تصحیح خطا لازم است؟

بنچمارک\footnote{\lr{benchmark}. معادل فارسی هنوز در جامعه تثبیت نشده است.} در این سند یعنی مقایسه کنترل‌شده با یک مرجع، نه عدد تبلیغاتی.

\section{حداقل فرمالیسم}

زنجیره:
\begin{equation}
	\text{حالت / مدار}
	\ \longrightarrow\
	\text{خوانش}
	\ \longrightarrow\
	\widehat{\langle O\rangle}
	\ \longrightarrow\
	\text{مرجع کلاسیک}
	\ \longrightarrow\
	\text{هزینه }(n,d,t,\varepsilon).
\end{equation}
برون‌یابی به نویز صفر و لغو احتمالاتی خطا دو نمونه از سرکوب خطا هستند؛ یادگیری نویز پیش‌نیاز بعضی از آن‌هاست~\cite{Endo2021}. هدف نسخه اول فقط این است که این‌ها \emph{بعد از} اندازه‌گیری می‌آیند، نه به‌جای الگوریتم.
