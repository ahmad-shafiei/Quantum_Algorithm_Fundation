%--------------------------------------------------------
\begin{frame}{Classical Hamiltonians}
	\begin{itemize}
		\item Generalized coordinates and momenta $(q,p)$
		\item Legendre transform from the Lagrangian to $H$
		\item For usual conserved systems, $H$ is the total energy
	\end{itemize}
	\begin{equation*}
		\dot q=\frac{\partial H}{\partial p},\qquad
		\dot p=-\frac{\partial H}{\partial q}
	\end{equation*}
	\outlinebox{$H$ is both the energy function and the generator of classical motion.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Two Roles of the Classical $H$}
	\begin{enumerate}
		\item Energy on phase space
		\item Generator of classical time evolution
	\end{enumerate}
	\begin{alertblock}{Important distinction}
		Minimizing energy (equilibria / optimization) is not the same as following dynamics.
	\end{alertblock}
	\outlinebox{Minimization maps to combinatorial Ising costs; dynamics maps to trajectories.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Quantum Hamiltonians}
	\begin{itemize}
		\item Canonical quantization: $[q,p]=i\hbar$
		\item Dynamics: $i\hbar\,\partial_t\lvert\psi\rangle=\hat H\lvert\psi\rangle$
		\item Spectrum: $\hat H\lvert E_n\rangle=E_n\lvert E_n\rangle$
	\end{itemize}
	\outlinebox{$H$ both sets the allowed energies and generates the evolution of the state.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Why $H$ Is Central in Simulation}
	\begin{itemize}
		\item Almost every task is a function of one Hamiltonian:
		\begin{itemize}
			\item time evolution $e^{-iHt}$
			\item ground state and spectrum
			\item scheduled path $H(s)$
			\item more general $f(H)$
		\end{itemize}
		\item Choosing Trotter or VQE before fixing $H$ is premature
	\end{itemize}
	\outlinebox{Fix the model and $H$ first; then choose the question and the method.}
\end{frame}

\begin{frame}{Problem Classes in Quantum Simulation}
	\begin{itemize}
		\item Dynamic: post-quench response, transport, observable evolution
		\item Static: ground energies, optimal Ising configurations
		\item Intermediate: low-lying spectrum, gaps, state preparation
	\end{itemize}
	\outlinebox{Classify the question before choosing a method.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Two Main Approaches}
	\begin{enumerate}
		\item \textbf{Time evolution:} build or approximate $e^{-iHt}$ or $f(H)$
		\item \textbf{Adiabatic:} slow path $H_B\to H_P$, stay in the instantaneous ground state
	\end{enumerate}
	\begin{block}{Focus ahead}
		Adiabatic computation, worked example, Qiskit outline, then gap, runtime, and complexity.
	\end{block}
	\outlinebox{Trotter, LCU, and QSP are named only as tools for the evolution approach.}
\end{frame}

%========================================================
\section{Adiabatic computation}
\begin{frame}{Adiabatic Idea in One Slide}
	\begin{enumerate}
		\item Start from an easy $H_B$ with a known ground state
		\item Interpolate slowly to the problem Hamiltonian $H_P$
		\item Sufficient gap and time $\Rightarrow$ ground state of $H_P$
	\end{enumerate}
	\outlinebox{Encode the problem in a Hamiltonian path rather than enumerating configurations.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Problem Scope for Adiabatic Methods}
	\begin{enumerate}
		\item Combinatorial optimization mapped to classical Ising
		\item Ground-state preparation of physical models
		\item Hardware that natively implements a schedule $H(s)$ (annealing / analog)
	\end{enumerate}
	\outlinebox{Short examples: MaxCut, transverse-field Ising, analog annealers.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Why Use Adiabatic Computation?}
	\begin{itemize}
		\item Transparent physical picture of the computation
		\item Natural match to analog field schedules
		\item Digitally expressible via Trotterized paths (digital adiabatic)
	\end{itemize}
	\outlinebox{Less natural when the target is a precise $e^{-iHt}$ or a general $f(H)$.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{The Path $H(s)$}
	\begin{itemize}
		\item Standard linear interpolation
		\[
			H(s)=(1-s)\,H_B + s\,H_P,\qquad 0\le s\le 1
		\]
		\item Example schedule: $s(t)=t/T$
		\item $s$ is not physical time; it mixes the two Hamiltonians
	\end{itemize}
	\outlinebox{$H(s)$ is linear in $s$; the eigenvalues generally are not.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Instantaneous Gap and Runtime $T$}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{itemize}
				\item $\Delta(s)=E_1(s)-E_0(s)$
				\item $\Delta_{\min}=\min_s\Delta(s)$ is the bottleneck
				\item Heuristic scale: $T\propto 1/\Delta_{\min}^{2}$
			\end{itemize}
		\end{column}
		\begin{column}{0.50\textwidth}
			\centering
			\begin{tikzpicture}[x=0.82cm, y=0.58cm]

\section{Worked example}
\begin{frame}{Two-Qubit Setup}
	\begin{itemize}
		\item Make $H_B$, $H_P$, the gap, and $T$ concrete
		\item System: two spin-$1/2$ particles (two qubits)
		\item Parameters: $h_x=1$, $J=1$, $h_z=1/4$
	\end{itemize}
	\outlinebox{The example is small enough that the full spectrum can be seen numerically.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Driver Hamiltonian $H_B$}
	\begin{itemize}
		\item $H_B=-h_x(X_1+X_2)$
		\item Easy ground state $\lvert++\rangle$, prepared by Hadamards on $\lvert00\rangle$
		\item Role: a known, preparable starting point
	\end{itemize}
	\outlinebox{Transverse field; product eigenstates of $X$.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Problem Hamiltonian $H_P$}
	\begin{itemize}
		\item $H_P=-J Z_1Z_2 - h_z(Z_1+Z_2)$
		\item Target ground state $\lvert00\rangle$ for $J,h_z>0$
		\item Ferromagnetic coupling plus a longitudinal field that lifts the degeneracy
	\end{itemize}
	\outlinebox{Optimization reading: the ground configuration is the solution.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Numerical Path Result}
	\begin{itemize}
		\item Gap minimum near $s_*\simeq 0.62$
		\item $\Delta_{\min}\simeq 0.66$
		\item Endpoint gaps alone do not locate the bottleneck
	\end{itemize}
	\outlinebox{Scan the instantaneous spectrum along the whole path $H(s)$.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{From Gap to a Runtime Estimate}
	\begin{itemize}
		\item $\lVert H_P-H_B\rVert_2 \simeq 2.37$
		\item Heuristic with $\varepsilon\sim 10^{-2}$: $T_{\mathrm{heur}}\sim 5\times 10^{2}$
		\item For $n=2$, $T\sim 40$ already yields near-unit success probability
	\end{itemize}
	\outlinebox{Distinguish a heuristic bound from a practical runtime checked by dynamics.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Code Steps (1): Hamiltonians and Gap}
	\begin{enumerate}
		\item Build $H_B$ and $H_P$ with \texttt{SparsePauliOp}
		\item Form $H(s)=(1-s)H_B+sH_P$

%========================================================
\section{Challenges}
\begin{frame}{Gap Is Not a Single Number}
	\begin{itemize}
		\item Full gap versus the gap inside the reachable symmetry sector
		\item Avoided crossing versus a true level crossing
		\item Symmetry-breaking noise can remove protection
	\end{itemize}
	\outlinebox{The smallest spectral distance alone is not a sufficient criterion.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Three Meanings of Required Time}
	\begin{enumerate}
		\item Heuristic / order-of-magnitude estimate
		\item Sufficient bound under explicit assumptions
		\item Practical time from dynamics and a success threshold
	\end{enumerate}
	\outlinebox{These three numbers usually disagree. Local scheduling: slow down near the bottleneck.}
\end{frame}

%========================================================
\section{Complexity}
\begin{frame}{Scaling with System Size}
	\begin{itemize}
		\item Polynomial gap closing: $\Delta_{\min}(n)\sim n^{-\alpha}$
		\item Exponential closing: $\Delta_{\min}(n)\sim e^{-cn}$
		\item With $T\sim 1/\Delta_{\min}^{2}$, the runtime may become exponential
	\end{itemize}
	\outlinebox{Qubit count alone is not a measure of difficulty.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Milder vs.\ Hard Regimes}
	\begin{columns}[T]
		\begin{column}{0.48\textwidth}
			\begin{block}{Milder}
				\begin{itemize}
					\item Gapped paths
					\item Away from criticality
					\item Some solvable models
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.48\textwidth}
			\begin{alertblock}{Hard}
				\begin{itemize}
					\item Near critical points
					\item Tunneling / exponentially small gaps
					\item Generic local ground-energy problems
				\end{itemize}
			\end{alertblock}
		\end{column}
	\end{columns}
	\vspace{0.3em}
	\outlinebox{In the worst case, estimating the ground energy of a local Hamiltonian is QMA-complete.}
\end{frame}

%--------------------------------------------------------
\begin{frame}{Three Complexity Layers}
	\begin{enumerate}
		\item Hardness of the \textbf{ground-state problem}
		\item Hardness of a \textbf{particular adiabatic path}
		\item Cost of a \textbf{digital implementation} of that path
	\end{enumerate}
	\begin{alertblock}{Message}
		Keep these three layers separate when assessing feasibility.
	\end{alertblock}
\end{frame}

%========================================================
\section{Summary}
\begin{frame}{Key Takeaways}
	\begin{enumerate}
		\item Simulation starts by fixing $H$ and the degrees of freedom
		\item Static and dynamic questions are distinct for the same $H$
		\item Adiabatic computation is a path $H(s)$; the gap sets the time
		\item Complexity follows $\Delta_{\min}(n)$ and the problem class, not $n$ alone
	\end{enumerate}
\end{frame}

	\end{itemize}
